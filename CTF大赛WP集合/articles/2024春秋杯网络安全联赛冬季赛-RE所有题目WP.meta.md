# 2024 春秋杯冬季赛 RE 所有题目 WP

## 题目 1：custom_md5_init + srand

```c
unsigned __int64 main_program() {
    unsigned int seed[2]; __int64 v2;
    char s[48]; char s1[56];
    *(_QWORD *)seed = 0LL; v2 = 0LL;
    custom_md5_init(seed);                       // 自实现 MD5
    srand(seed[0]);
    fgets(s, 43, stdin);
    if (strlen(s) == 42) {
        memset(s1, 0, 0x2B);
        xor_string_with_rand(s, s1);
        if (!memcmp(s1, &ida_chars, 0x2B)) puts("right");
    }
}
void xor_string_with_rand(__int64 a1, __int64 a2) {
    for (int i = 0; i <= 41; ++i)
        *(_BYTE *)(i + a2) = (rand() % 127) ^ *(_BYTE *)(i + a1);
}
unsigned __int64 custom_md5_init(__int64 a1) {
    char v2[96]; char v3[1032];
    qmemcpy(v3, custom_md5_init, 0x400uLL);   // 拷自身 1024 字节
    MD5_Init(v2, (char *)custom_md5_init + 1024, 128LL);
    MD5_Update(v2, v3, 1024LL);
    MD5_Final(a1, v2);
}
```

`custom_md5_init` 把函数自身二进制 1024 字节做 MD5_Update → `seed[0]` → `srand(seed[0])` → `rand()%127` 异或 input。  
IDA 数据段 `ida_chars` 0x2B 字节已知 → 直接 `xor[i] ^ cmp[i]` 还原。

```python
xor = [58, 26, 43, 31, 110, 0, 50, 69, 82, 68, 34, 85, 58, 55, 125, 67, 123, 35, 82, 28, 96, 70, 10, 7, 86, 56, 114, 121, 16, 29, 82, 74, 47, 117, 97, 22, 117, 20, 92, 65, 88, 118]

for i in range(42):
    print(chr(xor[i] ^ cmp[i]), end="")
```
