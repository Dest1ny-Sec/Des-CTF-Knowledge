nbit = 128

class LFSR:
    def __init__(self, seed: int, mask: int):
        self.state = seed & (2**nbit - 1)
        self.mask = mask & (2**nbit - 1)

    def __next__(self):
        b = (self.state & self.mask).bit_count() & 1
        self.state = ((self.state << 1) | b) & (2**nbit - 1)
        return b
    
    def __call__(self, bits: int):
        out = 0
        for _ in range(bits):
            out = (out << 1) | next(self)
        return out

class NFSR:
    def __init__(self, lfsr0: LFSR, lfsr1: LFSR, noise: iter):
        self.lfsr0 = lfsr0
        self.lfsr1 = lfsr1
        self.noise = noise
    
    def __next__(self):
        b0, b1 = self.lfsr0(1), self.lfsr1(1)
        b = b0 ^ b1 ^ next(self.noise)
        return b
    
    def __call__(self, bits: int):
        out = 0
        for _ in range(bits):
            out = (out << 1) | next(self)
        return out

    def encrypt(self, msg: bytes):
        return bytes([m ^ self(8) for m in msg])

def NOISE(lfsr: LFSR, p: float=2/3, prec: int=256):
    import secrets
    
    t = 2**prec
    while True:
        if lfsr(1):
            yield int(secrets.randbelow(t) / t > p)
        else:
            yield int(secrets.randbelow(t) / t <= p)

目前公开的情报：

密钥流输出收到两个 LFSR 和 一个 NOISE 控制，output = lfsr1^lfsr2^noise

极限可以获得 65536 个字节密钥流，但是可以选择：交互 512 次，每次搞 128 字节；交互 256 次，每次搞 256 字节；交互 128 次，每次搞 512 字节；（不过肯定得留一次交互去搞 flag）

每一次交互，两个 LFSR 的 seed 都会重置，但是 noise 不会，因此如果不存在这个 noise，每一次的 output 都是固定不变的。

noise 的输出受独立的一个 lfsrn 影响，当 lfsrn 输出 1 时，noise 有 1/3 的概率为 1；当 lfsrn 输出 0 时，noise 有 2/3 的概率为 1；

lfsrn 的 secret 和 mask 只有 12 比特，所以它的周期最大就是 2**12 = 4096比特=512字节
