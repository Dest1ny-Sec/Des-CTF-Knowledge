## 4 种迷宫

### 1. 10x10 数组迷宫
```c
char maze[SIZE][SIZE] = {
    {'S', '#', '#', ...},  // S 在 (0,0)
    {'#', '*', '*', ...},  // E 在 (5,7)
    ...
};
scanf(" %c", &move);
switch(move) {
    case 'w': if (x > 0 && maze[x-1][y] != '#') x--; break;
    case 'a': if (y > 0 && maze[x][y-1] != '#') y--; break;
    case 's': if (x < SIZE-1 && maze[x+1][y] != '#') x++; break;
    case 'd': if (y < SIZE-1 && maze[x][y+1] != '#') y++; break;
}
```

### 2. 100 字节扁平迷宫
```c
char maze[100] = { '#', 'S', '#', ... };
// t 偏移
case 'w': t -= 10; break;
case 'a': t--; break;
case 's': t += 10; break;
case 'd': t++; break;
```

### 3. ACTF W/E/M/J 校验迷宫
```c
_BOOL8 sub_78A(__int64 a1) {
    int v2 = 0, v3 = 5, v4 = 0;
    while (byte_201020[v2] != 0x21) {
        v2 -= v4;
        if (s[v3] == 'W' && v4 != -16) v4 = -16;  // W = 上移 -16
        if (s[v3] == 'E' && v4 != 1) v4 = 1;       // E = 右移 +1
        if (s[v3] == 'M' && v4 != 16) v4 = 16;     // M = 下移 +16
        if (s[v3] == 'J' && v4 != -1) v4 = -1;     // J = 左移 -1
        ++v3;
        while (!byte_201020[v2]) {
            if (v4 == -1 && (v2 & 0xF) == 0) return 0;  // 左边界
            if (v4 == 1 && v2 % 16 == 15) return 0;      // 右边界
            if (v4 == 16 && (v2 - 240) <= 0xF) return 0;  // 下边界
            if (v4 == -16 && (v2 + 15) <= 0x1E) return 0; // 上边界
            v2 += v4;
        }
    }
    return s[v3] == 125;  // '}'
}
```
