# 腾讯游戏安全大赛初赛题解

**来源**: ctfiot.com ID 115172

## Q1: 找明文信息（1分）
- 64 位 Windows10 运行 contest.exe
- 写 contest.txt 密文: `ImVkImx9JG12OGtlImV+`
- CE 打开, memory view 找一直变化的内存
- 找到 contest.txt 字符串
- 找访问该地址的指令, 发现 r10 读取的内存
- memory view 查找 7FF713657190 → 找到 base 表:
  ```
  QRSTUVWXYZabcdefABCDEFGHIJKLMNOPwxyz0123456789+/ghijklmnopqrstuv
  ```
- base64 解密得 `catchmeifyoucan` (明文 flag)

## Q2: 修改内存让密文改明文（2分）

### 分析
- Base64: 三原文对应四密文
- 第一密文: 原文字符右移两位查表
- 第四密文: `and 0x3f` 取最低 6 位
- 中间密文: 由第一+第二+第三原文组合

### 关键 PATCH
```c
// PATCH 1: 右移两位查表部分
// contest.exe+BA39~BA41 全部 NOP (0x90)

// PATCH 2: and 0x3f 最低 6 位部分
// contest.exe+B9FD~BA04 全部 NOP

// PATCH 3: 中间密文
// contest.exe+BA5D~BA68 全部 NOP

// PATCH 4: 不递增 r14 (避免密文长度)
// contest.exe+BA05 起始改为:
mov [r14], al
add r14, 0
```

### VirtualProtect inline hook 修复
```c
BYTE INS[] = {0x4C, 0x8B, 0xD1, 0xB8, 0x50};  // 原始 ZwProtectVirtualMemory 字节

// 遍历模块找 ZwProtectVirtualMemory
MODULEENTRY32 me;
if (addr = (UINT64)GetProcAddress(me.hModule, "ZwProtectVirtualMemory")) break;

// 修复 inline hook
VirtualProtectEx(hProcess, (void *)addr, 0x5, PAGE_EXECUTE_READWRITE, &old);
WriteProcessMemory(hProcess, (void *)addr, INS, 0x5, &written);
VirtualProtectEx(hProcess, (void*)addr, 0x5, old, &old);
```

### 远程线程注入
```c
void InjectModule(DWORD ProcessId, const char* szPath) {
    HANDLE hProcess = OpenProcess(PROCESS_ALL_ACCESS, FALSE, ProcessId);
    LPVOID lpAddress = VirtualAllocEx(hProcess, NULL, 0x100, MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);
    WriteProcessMemory(hProcess, lpAddress, szPath, strlen(szPath), &dwWriteLength);
    HANDLE hThread = CreateRemoteThread(hProcess, NULL, NULL,
        (LPTHREAD_START_ROUTINE)LoadLibraryA, lpAddress, NULL, NULL);
    WaitForSingleObject(hThread, -1);
    VirtualFreeEx(hProcess, lpAddress, 0, MEM_RELEASE);
}
```

### DllMain patch()
```c
UINT64 Offset[3] = {0xBA39, 0xB9FD, 0xBA5D};
UINT64 Len[3] = {9, 8, 12};
BYTE Ins[] = {
    0x41, 0x88, 0x06,  // mov [r14], al
    0x90,               // nop
    0x49, 0x83, 0xC6, 0x00  // add r14, 0
};
UINT64 InsOffset = 0xBA05;

void patch() {
    UINT64 Base = (UINT64)GetModuleHandle(nullptr);
    for (int i = 0; i < 3; i++) {
        UINT64 addr = Base + Offset[i];
        VirtualProtect((void *)addr, Len[i], PAGE_EXECUTE_READWRITE, &oldprot);
        memcpy((void *)addr, NOP, Len[i]);
        VirtualProtect((void*)addr, Len[i], oldprot, &oldprot);
    }
    VirtualProtect((void*)(Base + InsOffset), InsLen, PAGE_EXECUTE_READWRITE, &oldprot);
    memcpy((void *)(Base + InsOffset), Ins, InsLen);
    VirtualProtect((void*)(Base + InsOffset), InsLen, oldprot, &oldprot);
}
```
