**QR_Maze.xlsx(Excel转二维码)**
```python
import zipfile, xml.etree.ElementTree as ET, re
from PIL import Image
f = "QR_Maze.xlsx"

def a1rc(a1):
    m = re.match(r"([A-Z]+)(\d+)", a1)
    col = 0
    for ch in m.group(1):
        col = col*26 + (ord(ch)-64)
    return int(m.group(2)), col

def load(f):
    with zipfile.ZipFile(f,'r') as z:
        ws = [n for n in z.namelist() if n.startswith("xl/worksheets/sheet")]
        ws.sort()
        xml = z.read(ws[0])
        ss = []
        if "xl/sharedStrings.xml" in z.namelist():
            root = ET.fromstring(z.read("xl/sharedStrings.xml"))
            for si in root.findall(".//{...}si"):
                ss.append("".join([t.text or "" for t in si.findall(".//{...}t")]))
    ns = {"a":"http://schemas.openxmlformats.org/spreadsheetml/2006/main"}
    root = ET.fromstring(xml)
    cels = {}; mr = mc = 0
    for c in root.findall(".//a:c",ns):
        r = c.attrib.get("r"); t = c.attrib.get("t")
        rr,cc = a1rc(r)
        v = c.find("a:v",ns)
        val = 0
        if v is not None and v.text:
            if t=="s":
                s = ss[int(v.text)].strip()
                val = 1 if s not in ("","0") else 0
            else:
                try: val = 1 if float(v.text.strip())>0.5 else 0
                except: val = 0
        cels[(rr,cc)] = val; mr=max(mr,rr); mc=max(mc,cc)
    g = [[0]*mc for _ in range(mr)]
    for (rr,cc),v in cels.items(): g[rr-1][cc-1]=v
    return g

g = load(f)
h,w = len(g), len(g[0])
img = Image.new("L",(w,h))
px = img.load()
for y in range(h):
    for x in range(w):
        px[x,y] = 255 if g[y][x]==1 else 0
img = img.resize((w*6,h*6), Image.NEAREST)
img.save("1.png")
```
